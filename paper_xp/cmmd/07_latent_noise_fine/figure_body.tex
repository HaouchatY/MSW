% Compensating degradations, in the two latent-noise settings of the CMMD paper,
% with an 8-pixel VQGAN (VQ-Diffusion ITHQ, 64x64 tokens at 512 px -- the
% granularity of Muse's 512-px stage); a fraction p of the tokens replaced at
% random.  Left: COCO-30k against the clean images (CMMD Fig. 1).  Middle:
% Stable Diffusion 1.4 images against COCO-30k (CMMD Fig. 7).  Both: change
% from p = 0 in percent of each metric's own value.  Right: MSW's squared
% distance u, change split exactly by pyramid level (COCO setting).
% Data: data/comp_{coco,sd,levels}.csv.

  xticklabel style={/pgf/number format/fixed, /pgf/number format/precision=2},
  ymin=-12, ymax=95, legend style={at={(0.03,0.97)}, anchor=north west}]

\addplot[cfid] table[x=p, y=fid, col sep=comma]{data/comp_coco.csv};

\addplot[cours] table[x=p, y=msw, col sep=comma]{data/comp_coco.csv};

\addplot[fill=black!50, draw=black!70] table[x=level, y=d002, col sep=comma]{data/comp_levels.csv};
\addplot[fill=black!85, draw=black] table[x=level, y=d005, col sep=comma]{data/comp_levels.csv};
