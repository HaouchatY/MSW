% Shared preamble of every figure.tex: the paper's pgfplots style and a page
% cropped to the figure (no standalone class needed).
\documentclass{article}
\usepackage{times,amsmath}
\usepackage{tikz,pgfplots}
\usetikzlibrary{calc,arrows.meta,patterns}
\usepgfplotslibrary{groupplots,fillbetween}
\input{../../common/style}
\setlength\textwidth{17.46cm}\setlength\linewidth{\textwidth}   % the paper's text width
\newsavebox\figbox
\newcommand\cropfigure[1]{%
  \savebox\figbox{\input{#1}}%
  \pdfpagewidth=\dimexpr\wd\figbox+6pt\relax
  \pdfpageheight=\dimexpr\ht\figbox+\dp\figbox+6pt\relax
  \hoffset=-1in \voffset=-1in
  \shipout\hbox{\kern3pt\vbox{\kern3pt\box\figbox\kern3pt}}}
